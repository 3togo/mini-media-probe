cmake_minimum_required(VERSION 3.20)
project(mini_media_probe VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PROBE_STATIC_FFMPEG "Link FFmpeg and its private dependencies statically" ON)
option(PROBE_REQUIRE_DURATION_FIXTURES "Require ffmpeg/LAME/x264 for duration regression tests" OFF)
find_package(PkgConfig REQUIRED)
if(PROBE_STATIC_FFMPEG)
  set(PKG_CONFIG_ARGN --static)
  if(MINGW)
    set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
  endif()
endif()
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat>=60 libavcodec libavutil)
add_executable(mini-media-probe src/main.cpp)
target_link_libraries(mini-media-probe PRIVATE PkgConfig::AV)
if(NOT WIN32)
  target_compile_definitions(mini-media-probe PRIVATE _FILE_OFFSET_BITS=64)
endif()
if(MINGW)
  target_link_options(mini-media-probe PRIVATE -static)
endif()
install(TARGETS mini-media-probe RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/mini-media-probe)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/mini-media-probe)
install(DIRECTORY "${CMAKE_BINARY_DIR}/licenses/" DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/mini-media-probe OPTIONAL)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME helper COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_helper.py $<TARGET_FILE:mini-media-probe>)
  add_test(NAME ffprobe-normalisation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ffprobe_normalisation.py)
  set(DURATION_TEST_ARGS)
  if(PROBE_REQUIRE_DURATION_FIXTURES)
    list(APPEND DURATION_TEST_ARGS --require-fixtures)
  endif()
  add_test(NAME duration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_duration.py $<TARGET_FILE:mini-media-probe> ${DURATION_TEST_ARGS})
  set_tests_properties(helper PROPERTIES TIMEOUT 120)
  set_tests_properties(duration PROPERTIES TIMEOUT 120)
endif()
