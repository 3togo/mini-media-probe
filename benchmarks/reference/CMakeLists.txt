cmake_minimum_required(VERSION 3.20)
project(mediainfo_reference LANGUAGES CXX)
find_library(MEDIAINFO mediainfo PATHS "${MEDIA_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(ZEN zen PATHS "${MEDIA_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_executable(mediainfo-reference mediainfo.cpp)
target_compile_features(mediainfo-reference PRIVATE cxx_std_17)
target_compile_definitions(mediainfo-reference PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(mediainfo-reference PRIVATE ${MEDIAINFO} ${ZEN} ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS})
